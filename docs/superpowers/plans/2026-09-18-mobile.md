# Игра на телефоне — план реализации

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** та же игра по той же ссылке открывается на телефоне и играется двумя пальцами — стики, экранные кнопки, адаптивные панели, только горизонтальная ориентация.

**Architecture:** тач не заводит новую модель ввода, а заполняет то же состояние, которое уже читают игрок, камера и стрельба: левый стик даёт вектор движения, правый ставит синтетический курсор в `mouse.x/y`, отклонение за порог — это `mouse.down`. В боевом коде меняется одна строка (движение берёт вектор из `moveAxis()`); всё остальное — новый код рядом с существующим.

**Tech Stack:** один файл `index.html` (Canvas 2D, WebAudio, без зависимостей); тесты — `tests/suites/*.js`, собираются `tests/build.py`, запускаются `./tests/run.sh` на node + python3.

**Spec:** `docs/superpowers/specs/2026-09-18-mobile-design.md`

## Global Constraints

- Игра остаётся **одним файлом** `index.html` без внешних зависимостей и без сборки.
- Клавиатурный и мышиный путь не меняется: существующие 3110+ проверок обязаны остаться зелёными после каждой задачи.
- Весь текст интерфейса и комментарии в коде — по-русски, как во всём проекте.
- Каждая задача заканчивается зелёным `./tests/run.sh` и отдельным коммитом.
- Комментарии объясняют «почему», а не пересказывают код — стиль существующего файла.
- Числа из спецификации берутся дословно: `RADIUS 58`, `DEAD 0.18`, `WALK 0.45`, `FIRE 0.55`, `CURSOR 240`, конус доводчика `0.12` рад и дальность `520`, масштаб на тач не ниже `0.85`, смещение камеры `0.45`, `hudScale = clamp(min(view.w/1280, view.h/800), 0.62, 1)`, плотность пикселей на тач `1.5`, автокачество `260 → 130` лучей и `420 → 180` частиц.
- Прогресс на каждом устройстве свой: ничего не синхронизируем.
- Коммиты заканчиваются строкой `Co-Authored-By: Claude Opus 5 <noreply@anthropic.com>`.

/* ═══════════════════════════════════════════════════════════════════════════
   ВВОД С ЭКРАНА — тач не заводит вторую модель управления: он заполняет то же
   состояние, что мышь и клавиши. Левый стик даёт вектор движения, правый
   ставит курсор — дальше работает обычный код игрока и камеры.
   ═══════════════════════════════════════════════════════════════════════════ */
const STICK = {
  RADIUS: 58,      // ход стика до упора
  DEAD: 0.18,      // мёртвая зона: палец дрожит всегда
  WALK: 0.45,      // слабее — тихий шаг, сильнее — бег
  FIRE: 0.55,      // порог автоматического огня
  CURSOR: 240,     // на сколько синтетический курсор отходит от центра экрана
};

const input = {
  mode: 'kbm',                      // 'kbm' или 'touch', переключается по событиям
  moveX: 0, moveY: 0, walk: false,
  sticks: { move: null, aim: null },
  pointers: new Map(),              // id → к какому стику или кнопке привязан палец
};
const touchMode = () => input.mode === 'touch';

/** Вектор стика в долях хода: длина 0…1, мёртвая зона уже вычтена. */
function stickVector(s) {
  if (!s) return { x: 0, y: 0, len: 0 };
  const dx = s.x - s.x0, dy = s.y - s.y0;
  const raw = Math.hypot(dx, dy) / STICK.RADIUS;
  if (raw < STICK.DEAD) return { x: 0, y: 0, len: 0 };
  const len = Math.min(1, (raw - STICK.DEAD) / (1 - STICK.DEAD));
  const inv = 1 / (Math.hypot(dx, dy) || 1);
  return { x: dx * inv, y: dy * inv, len };
}

/** Куда идёт игрок: клавиши или левый стик. Единственная точка, где сходятся
 *  оба способа управления. */
function moveAxis() {
  if (touchMode()) {
    const v = stickVector(input.sticks.move);
    return { x: v.x * v.len, y: v.y * v.len, walk: v.len > 0 && v.len < STICK.WALK };
  }
  let x = (held('d', 'в', 'arrowright') ? 1 : 0) - (held('a', 'ф', 'arrowleft') ? 1 : 0);
  let y = (held('s', 'ы', 'arrowdown') ? 1 : 0) - (held('w', 'ц', 'arrowup') ? 1 : 0);
  const len = Math.hypot(x, y);
  if (len > 1) { x /= len; y /= len; }
  return { x, y, walk: held('shift') };
}

function setInputMode(mode) {
  if (input.mode === mode) return;
  input.mode = mode;
  if (mode === 'kbm') { input.sticks.move = input.sticks.aim = null; input.pointers.clear(); }
}

function touchStart(id, x, y) {
  setInputMode('touch');
  // Левая половина экрана — движение, правая — прицел. Стик появляется там,
  // где палец коснулся: фиксированный стик заставляет целиться вслепую.
  const side = x < view.w / 2 ? 'move' : 'aim';
  if (input.sticks[side]) return;           // этот стик уже держат
  input.sticks[side] = { id, x0: x, y0: y, x, y };
  input.pointers.set(id, side);
}

function touchMove(id, x, y) {
  const side = input.pointers.get(id);
  if (!side) return;
  const s = input.sticks[side];
  if (s && s.id === id) { s.x = x; s.y = y; }
}

function touchEnd(id) {
  const side = input.pointers.get(id);
  input.pointers.delete(id);
  if (side && input.sticks[side] && input.sticks[side].id === id) input.sticks[side] = null;
}

const pointerXY = e => {
  const r = canvas.getBoundingClientRect();
  return { x: e.clientX - r.left, y: e.clientY - r.top };
};

canvas.addEventListener('pointerdown', e => {
  if (e.pointerType === 'mouse') return;
  e.preventDefault();
  Audio2D.resume();
  const p = pointerXY(e);
  touchStart(e.pointerId, p.x, p.y);
});
canvas.addEventListener('pointermove', e => {
  if (e.pointerType === 'mouse') return;
  e.preventDefault();
  const p = pointerXY(e);
  touchMove(e.pointerId, p.x, p.y);
});
for (const type of ['pointerup', 'pointercancel'])
  canvas.addEventListener(type, e => {
    if (e.pointerType === 'mouse') return;
    touchEnd(e.pointerId);
  });
```

/** Мягкий доводчик: если в узком конусе есть видимый живой враг, прицел
 *  смещается к нему на половину разницы. Пальцем по фигуре в 21 пиксель
 *  иначе не попасть, но выбор цели остаётся за игроком. */
const AIM_ASSIST = { CONE: 0.12, RANGE: 520, PULL: 0.5 };

function aimAssist(ang) {
  let best = null, bestOff = AIM_ASSIST.CONE;
  for (const b of bots) {
    if (!b.alive) continue;
    const d = dist(player.x, player.y, b.x, b.y);
    if (d > AIM_ASSIST.RANGE) continue;
    if (!playerSees(b.x, b.y)) continue;
    const to = Math.atan2(b.y - player.y, b.x - player.x);
    const off = Math.abs(angDiff(to, ang));
    if (off < bestOff) { bestOff = off; best = to; }
  }
  return best === null ? ang : ang + angDiff(best, ang) * AIM_ASSIST.PULL;
}

/** Правый стик: ставит курсор и решает, идёт ли огонь. Вызывается раз в кадр
 *  до обновления игрока — дальше работает обычный код прицела и стрельбы. */
function applyAimStick() {
  if (!touchMode()) return;
  const v = stickVector(input.sticks.aim);
  if (!v.len) {
    input.firing = false;
    if (!input.sticks.aim) mouse.down = false;
    return;
  }
  const ang = aimAssist(Math.atan2(v.y, v.x));
  mouse.x = view.w / 2 + Math.cos(ang) * STICK.CURSOR;
  mouse.y = view.h / 2 + Math.sin(ang) * STICK.CURSOR;
  input.firing = v.len >= STICK.FIRE;
  mouse.down = input.firing;
}
```
